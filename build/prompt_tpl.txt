<script>
(function () {
  var D = %%DATA%%;
  var list = document.getElementById('list'), msg = document.getElementById('msg');
  var fpri = document.getElementById('fpri'), fdept = document.getElementById('fdept'), fcat = document.getElementById('fcat'), q = document.getElementById('q');
  var cats = []; D.forEach(function (r) { var k = r.dept + ' · ' + r.cat; if (cats.indexOf(k) < 0) cats.push(k); });
  cats.forEach(function (k) { var o = document.createElement('option'); o.value = k; o.textContent = k; fcat.appendChild(o); });
  if (!D.some(function (r) { return r.new; })) fpri.querySelector('[value="new"]').disabled = true;
  var shown = [];
  function el(t, c, x) { var e = document.createElement(t); if (c) e.className = c; if (x != null) e.textContent = x; return e; }
  function copy(text, label) {
    var done = function () { msg.textContent = label + ' copied.'; };
    if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(text).then(done, function () { fallback(text); });
    else fallback(text);
  }
  function fallback(text) { var ta = document.createElement('textarea'); ta.value = text; document.body.appendChild(ta); ta.select(); try { document.execCommand('copy'); msg.textContent = 'Copied.'; } catch (e) { msg.textContent = 'Select the text and copy it manually.'; } ta.remove(); }
  function render() {
    var term = q.value.trim().toLowerCase(), pri = fpri.value, dept = fdept.value, cat = fcat.value;
    shown = D.filter(function (r) {
      if (pri === '1' && r.pri !== 1) return false;
      if (pri === 'new' && !r.new) return false;
      if (dept && r.dept !== dept) return false;
      if (cat && (r.dept + ' · ' + r.cat) !== cat) return false;
      if (term && (r.name + ' ' + r.cat + ' ' + r.sub + ' ' + r.area).toLowerCase().indexOf(term) < 0) return false;
      return true;
    });
    document.getElementById('shown').textContent = shown.length;
    list.innerHTML = '';
    if (!shown.length) { list.appendChild(el('p', 'empty', 'No details match. Change a filter or clear the search.')); return; }
    var frag = document.createDocumentFragment();
    shown.forEach(function (r) {
      var row = el('article', 'row'), top = el('div', 'rtop'), h = el('h2', null, r.name), m = el('span', 'meta', r.dept + ' · ' + r.cat + ' · ' + r.sub);
      top.appendChild(h); top.appendChild(m); row.appendChild(top);
      var tags = el('div', 'ract');
      if (r.new) tags.appendChild(el('span', 'tag new', 'New'));
      tags.appendChild(el('span', 'tag' + (r.pri === 1 ? ' p1' : ''), r.pri === 1 ? 'Priority' : 'Later season'));
      tags.appendChild(el('span', 'tag', r.seasons.join(' · ')));
      row.appendChild(tags);
      row.appendChild(el('p', 'prompt', r.prompt));
      var act = el('div', 'ract');
      var b1 = el('button', 'btn small', 'Copy prompt'); b1.type = 'button'; b1.addEventListener('click', function () { copy(r.prompt, 'Prompt'); });
      var b2 = el('button', 'btn ghost small', 'Copy file name'); b2.type = 'button'; b2.addEventListener('click', function () { copy(r.file, 'File name'); });
      act.appendChild(b1); act.appendChild(b2); act.appendChild(el('span', 'file', 'Save as: ' + r.file));
      row.appendChild(act); frag.appendChild(row);
    });
    list.appendChild(frag);
  }
  [fpri, fdept, fcat].forEach(function (s) { s.addEventListener('change', render); });
  q.addEventListener('input', render);
  document.getElementById('copyall').addEventListener('click', function () {
    copy(shown.map(function (r) { return r.file + '\n' + r.prompt; }).join('\n\n'), shown.length + ' prompts');
  });
  var dl = null;
  function getDl() { if (!dl) dl = (window.claude && window.claude.use) ? window.claude.use('downloads').catch(function () { return null; }) : Promise.resolve(null); return dl; }
  document.getElementById('csv').addEventListener('click', function () {
    var esc = function (v) { return '"' + String(v).replace(/"/g, '""') + '"'; };
    var csv = ['File name,Detail,Department,Category,Subcategory,Seasons,Priority,Prompt'].concat(shown.map(function (r) { return [r.file, r.name, r.dept, r.cat, r.sub, r.seasons.join(' '), r.pri === 1 ? 'Priority' : 'Later', r.prompt].map(esc).join(','); })).join('\n');
    getDl().then(function (d) {
      if (!d) { msg.textContent = 'Downloads work when this page is open on claude.ai.'; return; }
      return d.save({ filename: 'forecut-image-prompts.csv', data: csv }).then(function () { msg.textContent = 'Saved forecut-image-prompts.csv.'; }, function (e) { msg.textContent = e && e.code === 'declined' ? 'Download cancelled.' : 'Downloads are not available in this view.'; });
    });
  });
  render();
})();
</script>
